! =============================================================
! SW1-USUARIOS (IOSvL2)
! Gi0/0 -> FortiGate port2 (trunk VLAN 10,20)
! Gi0/1 -> PC-VLAN10   | Gi0/2 -> PC-VLAN20
! =============================================================
enable
conf t
hostname SW1-USUARIOS
!
! --- VLANs ---
vlan 10
 name USUARIOS10
vlan 20
 name USUARIOS20
vlan 99
 name NATIVA
vlan 999
 name BLACKHOLE
exit
!
! --- Trunk hacia el FortiGate ---
interface Gi0/0
 description TRUNK-FORTIGATE-PORT2
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 99
 switchport trunk allowed vlan 10,20
 switchport nonegotiate
!
! --- Puertos de usuarios ---
interface Gi0/1
 description PC-VLAN10
 switchport mode access
 switchport access vlan 10
!
interface Gi0/2
 description PC-VLAN20
 switchport mode access
 switchport access vlan 20
!
interface range Gi0/1 - 2
 spanning-tree portfast
 spanning-tree bpduguard enable
 switchport port-security
 switchport port-security maximum 2
 switchport port-security mac-address sticky
 switchport port-security violation restrict
!
! --- Puertos sin uso ---
interface range Gi0/3, Gi1/0 - 3
 switchport mode access
 switchport access vlan 999
 shutdown
!
! --- DHCP Snooping ---
ip dhcp snooping
ip dhcp snooping vlan 10,20
no ip dhcp snooping information option
interface Gi0/0
 ip dhcp snooping trust
!
! --- Seguridad básica ---
enable secret Cisco123
service password-encryption
no ip domain-lookup
ip domain-name lab.local
username admin privilege 15 secret Cisco123
crypto key generate rsa modulus 2048
ip ssh version 2
line vty 0 4
 login local
 transport input ssh
line con 0
 login local
 exec-timeout 10 0
exit
banner motd #Acceso solo personal autorizado#
end
