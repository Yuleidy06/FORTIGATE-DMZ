# =============================================================
# FortiGate - Acceso inicial por consola
# Único paso por CLI: dar IP a VLAN20 para poder entrar al GUI.
# Todo lo demás se configuró y demostró por GUI.
# GUI: http://10.9.49.129 (desde PC-VLAN20)
# =============================================================
config system interface
 edit "VLAN20"
  set vdom "root"
  set interface "port2"
  set vlanid 20
  set ip 10.9.49.129 255.255.255.128
  set allowaccess ping https http
 next
end
